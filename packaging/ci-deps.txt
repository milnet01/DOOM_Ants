# Build dependencies for the Linux CI gate. SINGLE SOURCE OF TRUTH: both the
# GitHub workflow (.github/workflows/build.yml) and the local mirror
# (packaging/ci-local.sh) read this file, so the two can never drift.
#
# One deliberate exception: `freedoom` (the IWAD for the DOOM-0203 boot smoke) is
# installed by the smoke step itself in both places, not from this list -- it is a
# test fixture, not a build dependency, and the build+unit-test gate must not fail
# because a game-data package is unavailable.
#
# One apt (Debian/Ubuntu) package per line. Blank lines and lines whose first
# field starts with '#' are ignored; a trailing "# ..." comment on a package
# line is also ignored (only the first field is taken). Extract with:
#     awk 'NF && $1 !~ /^#/ {print $1}' packaging/ci-deps.txt
#
build-essential      # gcc/g++, make, libc headers
libsdl2-dev          # SDL2: window, input, audio device
libsdl2-mixer-dev    # SDL2_mixer: music + SFX
libvulkan-dev        # Vulkan loader + headers (3D back-end; linked even without a GPU)
glslc                # compiles the Vulkan shaders to SPIR-V
xxd                  # embeds the compiled shaders as C headers
# NB: fonts-dejavu-core used to be listed here because rb_text_test baked a system
# font and silently skipped itself when it found none. It now bakes the bundled
# assets/Oxanium-SemiBold.ttf (the same font the engine ships), so there is no host
# font dependency to install -- and no way for the test to skip itself. See
# DOOM-0243 / the 2026-07-26 test audit.
